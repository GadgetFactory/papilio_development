# PWB2 Integration Guide

This guide walks you through integrating Papilio Wishbone Bus 2 into your project.

## Prerequisites

- PlatformIO IDE or CLI
- ESP32 board
- FPGA with Wishbone-compatible peripherals
- SPI connection between ESP32 and FPGA

## Part 1: Project Setup

### 1. Create PlatformIO Project

```bash
pio project init --board esp32dev
```

### 2. Configure `platformio.ini`

```ini
[env:papilio]
platform = espressif32
board = esp32dev
framework = arduino

lib_deps =
    papilio_wishbone_bus_2
    # Add your peripheral libraries here
    pwb2_rgb_led_example

lib_extra_dirs =
    libs

extra_scripts = 
    pre:libs/papilio_wishbone_bus_2/tools/generate_pwb2_top.py

# SPI pin configuration
build_flags =
    -D PWB2_CS_PIN=5
    -D PWB2_MOSI_PIN=23
    -D PWB2_MISO_PIN=19
    -D PWB2_SCK_PIN=18
```

### 3. Directory Structure

Your project should look like:
```
my_project/
├── platformio.ini
├── src/
│   ├── main.cpp              # Your ESP32 code
│   └── gateware/             # Auto-generated
│       ├── top.v             # Generated by script
│       └── pwb2_config.vh    # Generated by script
└── libs/
    ├── papilio_wishbone_bus_2/
    └── (your peripheral libraries)
```

## Part 2: ESP32 Firmware

### Basic Example

```cpp
#include <Arduino.h>
#include <PapilioWishbone2.h>

// Create PWB2 instance
PapilioWishbone2 pwb2(PWB2_CS_PIN);

void setup() {
    Serial.begin(115200);
    
    // Initialize PWB2
    if (!pwb2.begin(20000000)) {  // 20 MHz SPI
        Serial.println("PWB2 init failed!");
        return;
    }
    
    // Discover connected peripherals
    pwb2.enumerateDevices();
    pwb2.printDevices(Serial);
    
    // Your initialization code
}

void loop() {
    // Your application code
}
```

### Finding Peripherals

```cpp
// Find by device ID
int findPeripheral(uint16_t deviceId) {
    for (int slot = 0; slot < 32; slot++) {
        if (pwb2.getDeviceId(slot) == deviceId) {
            return slot;
        }
    }
    return -1;
}

void setup() {
    // ...
    
    // Find RGB LED
    int rgbSlot = findPeripheral(0x4C45);
    if (rgbSlot >= 0) {
        Serial.printf("RGB LED found at slot %d\n", rgbSlot);
    }
}
```

### Writing to Peripherals

```cpp
// Example: Control RGB LED
void setColor(int slot, uint8_t r, uint8_t g, uint8_t b) {
    pwb2.writeSlot8(slot, 0, r);  // Red register
    pwb2.writeSlot8(slot, 1, g);  // Green register
    pwb2.writeSlot8(slot, 2, b);  // Blue register
}

// Example: Write to framebuffer
void updateFramebuffer(uint32_t* pixels, int count) {
    pwb2.writeExtendedBurst(0x2000, pixels, count);
}
```

## Part 3: FPGA Integration

### 1. Create a Peripheral

See the example in `examples/pwb2_rgb_led/pwb2_rgb_led.v`:

```verilog
`include "pwb2_pkg.vh"

module my_peripheral #(
    parameter SLOT_NUMBER = 1,
    parameter DEVICE_ID = 16'hXXXX
)(
    input wire clk,
    input wire rst,
    
    // Wishbone slave interface
    input wire [7:0] wb_adr_i,
    input wire [31:0] wb_dat_i,
    output reg [31:0] wb_dat_o,
    input wire wb_stb_i,
    input wire wb_we_i,
    output reg wb_ack_o,
    
    // Your peripheral ports
    output wire my_output
);

// Register map
reg [31:0] control_reg;
reg [31:0] data_reg;

// Wishbone interface
always @(posedge clk) begin
    if (rst) begin
        control_reg <= 32'h0;
        data_reg <= 32'h0;
        wb_ack_o <= 1'b0;
    end else begin
        wb_ack_o <= wb_stb_i && !wb_ack_o;
        
        if (wb_stb_i && !wb_ack_o) begin
            if (wb_we_i) begin
                // Write
                case (wb_adr_i)
                    8'h00: control_reg <= wb_dat_i;
                    8'h04: data_reg <= wb_dat_i;
                endcase
            end else begin
                // Read
                case (wb_adr_i)
                    8'h00: wb_dat_o <= control_reg;
                    8'h04: wb_dat_o <= data_reg;
                    default: wb_dat_o <= 32'h0;
                endcase
            end
        end
    end
end

// Your peripheral logic
assign my_output = control_reg[0];

endmodule
```

### 2. Create `wishbone_peripheral.json`

```json
{
  "name": "My Peripheral",
  "module_name": "my_peripheral",
  "device_id": 0x1234,
  "version": "1.0.0",
  "spi_transfer_bits": 32,
  "slots_required": 1,
  "instance_name": "my_periph",
  "description": "Description of what it does",
  "registers": [
    {
      "offset": 0,
      "name": "CONTROL",
      "width": 32,
      "access": "RW",
      "description": "Control register"
    },
    {
      "offset": 4,
      "name": "DATA",
      "width": 32,
      "access": "RW",
      "description": "Data register"
    }
  ],
  "hdl_files": [
    "my_peripheral.v"
  ],
  "ports": [
    {
      "name": "my_output",
      "direction": "output",
      "width": 1,
      "connection": "fpga_pin_signal",
      "description": "Output signal"
    }
  ]
}
```

### 3. Create `library.json`

```json
{
  "name": "my_pwb2_peripheral",
  "version": "1.0.0",
  "description": "My PWB2 peripheral",
  "keywords": ["pwb2", "wishbone", "peripheral"],
  "license": "MIT",
  "export": {
    "include": [
      "my_peripheral.v",
      "wishbone_peripheral.json"
    ]
  }
}
```

### 4. Add to Your Project

Place your peripheral in `libs/my_pwb2_peripheral/` and update `platformio.ini`:

```ini
lib_deps =
    papilio_wishbone_bus_2
    my_pwb2_peripheral
```

### 5. Build

Run PlatformIO build. The script will automatically:
1. Scan your libraries
2. Find all PWB2 peripherals
3. Generate `src/gateware/top.v`
4. Generate `src/gateware/pwb2_config.vh`

The generated `top.v` will include:
- PWB2 SPI bridge
- PWB2 interconnect
- Memory map (Slot 0)
- All your peripherals properly connected

## Part 4: Advanced Usage

### Burst Transfers

```cpp
// Fast framebuffer update
uint32_t framebuffer[320*240];

// Fill framebuffer
for (int i = 0; i < 320*240; i++) {
    framebuffer[i] = 0xFF0000;  // Red
}

// Upload with burst (much faster than single writes)
pwb2.writeExtendedBurst(0x2000, framebuffer, 320*240);
```

### Performance Monitoring

```cpp
void setup() {
    pwb2.begin();
    pwb2.resetStatistics();
}

void loop() {
    // Do lots of transactions
    
    Serial.printf("Transactions: %d\n", pwb2.getTransactionCount());
}
```

### Debug Mode

```cpp
void setup() {
    pwb2.begin();
    pwb2.setDebug(true);  // Enable debug output
    
    // Now all transactions will be logged
    pwb2.writeSlot8(1, 0, 0xFF);
    // Prints: [PWB2] Write slot 1 reg 0: 0xFF
}
```

### Multiple Slot Reads

```cpp
// Read multiple registers efficiently
void readAllSensors() {
    uint32_t data[4];
    pwb2.readSlotBurst(sensorSlot, 0, data, 4);
    
    int temp = data[0];
    int humidity = data[1];
    int pressure = data[2];
    int light = data[3];
}
```

## Part 5: Pin Connections

### ESP32 to FPGA SPI

```
ESP32          FPGA
-----          ----
GPIO 23 -----> MOSI
GPIO 19 <----- MISO
GPIO 18 -----> SCLK
GPIO 5  -----> CS_N
GND     -----> GND
```

Adjust pin numbers in `platformio.ini` as needed.

## Part 6: Troubleshooting

### Problem: "PWB2 init failed"
- Check SPI connections
- Verify FPGA is programmed and running
- Check SPI clock speed (try lower speed)
- Verify CS pin is correct

### Problem: "No peripherals found"
- Check that your peripherals have correct `wishbone_peripheral.json`
- Verify keywords include "pwb2" or "wishbone"
- Check that `lib_deps` includes your peripheral library
- Look at generated `top.v` to see if peripheral was included

### Problem: "Wrong data read back"
- Verify endianness (PWB2 uses MSB first)
- Check register addresses in peripheral
- Use debug mode to trace transactions
- Verify peripheral ACKs correctly

### Problem: "Generation script fails"
- Check Python is installed
- Verify `extra_scripts` path is correct
- Check for JSON syntax errors in metadata files
- Look at script output for specific error

## Part 7: Best Practices

### 1. Register Organization
```verilog
// Use 32-bit aligned addresses
0x00: CONFIG     (32-bit)
0x04: STATUS     (32-bit)
0x08: DATA       (32-bit)
0x0C: COUNT      (32-bit)
// Not: 0x00, 0x01, 0x02... (8-bit aligned)
```

### 2. Use Appropriate Tier
- **Slot mode**: Control registers, status, small data
- **Extended mode**: Framebuffers, audio buffers (up to 56KB)
- **Large mode**: External DDR for big datasets

### 3. Burst When Possible
```cpp
// Bad: 1000 single writes
for (int i = 0; i < 1000; i++) {
    pwb2.writeExtended(0x2000 + i*4, data[i]);
}

// Good: One burst write
pwb2.writeExtendedBurst(0x2000, data, 1000);
```

### 4. Check Capabilities
```cpp
uint32_t caps = pwb2.readSlot32(0, 0x04);
bool has_burst = caps & 0x01;
bool has_extended = caps & 0x02;
bool has_large = caps & 0x04;
```

### 5. Error Handling
```cpp
if (!pwb2.begin()) {
    Serial.println("Failed to initialize PWB2");
    // Handle error
    return;
}

if (pwb2.getSlotCount() == 0) {
    Serial.println("No peripherals detected");
    // Handle error
    return;
}
```

## Part 8: Example Projects

### Simple RGB LED Project

**platformio.ini:**
```ini
[env:papilio]
platform = espressif32
board = esp32dev
framework = arduino
lib_deps =
    papilio_wishbone_bus_2
    pwb2_rgb_led_example
lib_extra_dirs = libs
extra_scripts = pre:libs/papilio_wishbone_bus_2/tools/generate_pwb2_top.py
```

**src/main.cpp:**
```cpp
#include <PapilioWishbone2.h>

PapilioWishbone2 pwb2;
int rgbSlot = -1;

void setup() {
    Serial.begin(115200);
    pwb2.begin();
    pwb2.enumerateDevices();
    
    rgbSlot = findPeripheral(0x4C45);
}

void loop() {
    static uint8_t hue = 0;
    
    uint8_t r = (hue < 85) ? 255 - hue * 3 : 0;
    uint8_t g = (hue < 85) ? hue * 3 : (hue < 170) ? 255 : 0;
    uint8_t b = (hue >= 170) ? (hue - 170) * 3 : 0;
    
    pwb2.writeSlot8(rgbSlot, 0, r);
    pwb2.writeSlot8(rgbSlot, 1, g);
    pwb2.writeSlot8(rgbSlot, 2, b);
    
    hue++;
    delay(10);
}
```

That's it! The FPGA top-level is auto-generated.
